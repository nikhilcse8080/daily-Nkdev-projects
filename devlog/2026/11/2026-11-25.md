# 2026-11-25 — Parts catalog search API with a Next.js route handler

**Stack:** Next.js · **Topic:** App Router route handlers with query validation

A two-wheeler parts catalog needs a fast lookup by bike model and part name for the counter app. A single App Router route handler can validate query params, filter, paginate, and return JSON with cache headers.

```ts
// app/api/parts/route.ts
import { NextRequest, NextResponse } from "next/server";

type Part = { sku: string; name: string; models: string[]; price: number; stock: number };

const PARTS: Part[] = [
  { sku: "BRK-PAD-110", name: "Front brake pad", models: ["Activa 6G", "Jupiter"], price: 245, stock: 40 },
  { sku: "CHN-KIT-150", name: "Chain sprocket kit", models: ["Pulsar 150", "Apache RTR 160"], price: 1350, stock: 6 },
  { sku: "AIR-FLT-125", name: "Air filter", models: ["Shine 125", "Activa 6G"], price: 190, stock: 0 },
];

export async function GET(req: NextRequest) {
  const sp = req.nextUrl.searchParams;
  const q = (sp.get("q") ?? "").trim().toLowerCase();
  const model = (sp.get("model") ?? "").trim().toLowerCase();
  const page = Math.max(1, Number.parseInt(sp.get("page") ?? "1", 10) || 1);
  const limit = Math.min(50, Math.max(1, Number.parseInt(sp.get("limit") ?? "20", 10) || 20));

  if (q.length > 0 && q.length < 2) {
    return NextResponse.json({ error: "q must be at least 2 characters" }, { status: 400 });
  }

  const matches = PARTS.filter(
    (p) =>
      (!q || p.name.toLowerCase().includes(q) || p.sku.toLowerCase().includes(q)) &&
      (!model || p.models.some((m) => m.toLowerCase() === model))
  );

  const items = matches.slice((page - 1) * limit, page * limit);
  return NextResponse.json(
    { total: matches.length, page, limit, items },
    { headers: { "Cache-Control": "public, s-maxage=60, stale-while-revalidate=300" } }
  );
}
```

**Takeaway:** Clamp and validate every query param in the route handler itself — `?limit=10000` should never reach your data layer.
